  If you used the default placeholder secret.yaml, the application WILL NOT
  start correctly. Provide real secrets before deploying to production:

  kubectl create secret generic {{ include "audit-log-analyzer.secretName" . }} \
    --namespace {{ .Release.Namespace }} \
    --from-literal=secret-key=$(openssl rand -hex 32) \
    --from-literal=database-url='postgresql+asyncpg://...' \
    --from-literal=valkey-url='redis://:password@valkey:6379/0' \
    --from-literal=github-client-id='...' \
    --from-literal=github-client-secret='...' \
    --from-literal=github-rules-token=''  # optional — leave empty to disable rule sync \
    --from-literal=minio-root-user='...' \
    --from-literal=minio-root-password='...' \
    --from-literal=minio-ingest-user='...' \
    --from-literal=minio-ingest-password='...' \
    --dry-run=client -o yaml | kubectl apply -f -
